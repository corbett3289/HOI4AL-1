coi_alien_gateway_category = {
	coi_alien_construct_energy_gateway = {
		icon = generic_construction
		allowed = { original_tag = XAC }
		state_target = any_owned_state
		on_map_mode = map_and_decisions_view
		target_root_trigger = {
			has_tech = coi_alien_energy_gateway_synchronization
			check_variable = { coi_alien_gateway_committed_count < 3 }
		}
		target_trigger = { FROM = { coi_alien_gateway_valid_construction_state_trigger = yes } }
		visible = {
			has_tech = coi_alien_energy_gateway_synchronization
			check_variable = { coi_alien_gateway_committed_count < 3 }
			FROM = { coi_alien_gateway_valid_construction_state_trigger = yes }
		}
		available = {
			has_idea = coi_alien_mothership_anchorage_online
			check_variable = { coi_alien_gateway_committed_count < 3 }
			command_power > 24.9
			FROM = { coi_alien_gateway_valid_construction_state_trigger = yes }
		}
		cost = 75
		custom_cost_trigger = { command_power > 24.9 }
		custom_cost_text = coi_alien_gateway_cost_75pp_25cp
		days_remove = 120
		cancel_if_not_visible = no
		cancel_trigger = {
			OR = {
				NOT = { has_idea = coi_alien_mothership_anchorage_online }
				FROM = {
					OR = {
						NOT = { is_owned_by = ROOT }
						NOT = { is_controlled_by = ROOT }
					}
				}
			}
		}
		complete_effect = {
			add_command_power = -25
			add_to_variable = { coi_alien_gateway_committed_count = 1 }
			clamp_variable = { var = coi_alien_gateway_committed_count min = 0 max = 3 }
			FROM = { set_state_flag = coi_alien_gateway_construction_in_progress }
		}
		remove_effect = {
			if = {
				limit = {
					has_idea = coi_alien_mothership_anchorage_online
					FROM = {
						is_owned_by = ROOT
						is_controlled_by = ROOT
						NOT = { coi_alien_energy_gateway > 0 }
					}
				}
				FROM = {
					add_building_construction = {
						type = coi_alien_energy_gateway
						level = 1
						instant_build = yes
					}
					clr_state_flag = coi_alien_gateway_construction_in_progress
					set_state_flag = coi_alien_gateway_established
				}
				coi_alien_reconcile_gateway_committed_count_effect = yes
				coi_alien_recalculate_gateway_output_effect = yes
			}
			else = {
				FROM = { clr_state_flag = coi_alien_gateway_construction_in_progress }
				coi_alien_reconcile_gateway_committed_count_effect = yes
				add_political_power = 75
				add_command_power = 25
			}
		}
		cancel_effect = {
			FROM = { clr_state_flag = coi_alien_gateway_construction_in_progress }
			coi_alien_reconcile_gateway_committed_count_effect = yes
			add_political_power = 75
			add_command_power = 25
		}
		ai_will_do = {
			base = 20
			modifier = { factor = 0 check_variable = { coi_alien_gateway_committed_count > 2 } }
		}
	}
