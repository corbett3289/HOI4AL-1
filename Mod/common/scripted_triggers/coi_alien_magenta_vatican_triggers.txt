### Country- and state-scope gates for the Magenta / Vatican flavor layer.

coi_alien_magenta_german_alignment_trigger = {
	tag = ITA
	country_exists = GER
	has_country_flag = coi_alien_rs33_black_program
	NOT = { has_country_flag = coi_alien_magenta_german_sharing_resolved }
	NOT = { has_country_flag = coi_alien_magenta_german_sharing_prompted }
	GER = { NOT = { has_war_with = ITA } }
	OR = {
		is_in_faction_with = GER
		has_completed_focus = ITA_pact_of_steel
	}
}

coi_alien_has_any_disclosure_stance_trigger = {
	OR = {
		has_country_flag = coi_alien_disclosure_stance_secrecy
		has_country_flag = coi_alien_disclosure_stance_managed
		has_country_flag = coi_alien_disclosure_stance_international
		has_country_flag = coi_alien_disclosure_stance_open_files
	}
}

coi_alien_controls_magenta_vault_trigger = {
	controls_state = 159
	159 = { has_state_flag = coi_alien_magenta_craft_vault }
}

coi_alien_vatican_reckoning_host_trigger = {
	has_global_flag = coi_alien_catastrophic_disclosure_active
	has_global_flag = coi_alien_vatican_archive_sealed
	NOT = { has_global_flag = coi_alien_vatican_reckoning_prompted }
	NOT = { has_global_flag = coi_alien_vatican_reckoning_resolved }
	OR = {
		AND = {
			tag = PAP
			exists = yes
		}
		AND = {
			NOT = { country_exists = PAP }
			controls_state = 2
		}
	}
}
