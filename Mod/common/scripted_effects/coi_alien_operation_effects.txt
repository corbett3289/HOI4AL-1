coi_alien_complete_introduce_cascading_faults_effect = {
	ROOT = {
		remove_operation_token = {
			tag = FROM
			token = coi_alien_token_institutional_access
		}
		if = {
			limit = {
				has_operation_token = {
					tag = FROM
					token = coi_alien_token_research_access
				}
			}
			remove_operation_token = {
				tag = FROM
				token = coi_alien_token_research_access
			}
		}
	}

	FROM.FROM = {
		damage_building = {
			type = infrastructure
			damage = 1
			repair_speed_modifier = -0.25
		}

		# Priority order makes the result deterministic when several nodes exist.
		if = {
			limit = { radar_station > 0 }
			damage_building = {
				type = radar_station
				damage = 1
				repair_speed_modifier = -0.25
			}
		}
		else_if = {
			limit = { air_base > 0 }
			damage_building = {
				type = air_base
				damage = 1
				repair_speed_modifier = -0.25
			}
		}
		else_if = {
			limit = { naval_base > 0 }
			damage_building = {
				type = naval_base
				damage = 1
				repair_speed_modifier = -0.25
			}
		}
	}

	FROM = {
		add_timed_idea = {
			idea = coi_alien_cascading_faults
			days = 90
		}
		coi_alien_raise_awareness_one_effect = yes
	}
}
