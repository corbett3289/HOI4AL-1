		}
	}

	option = {
		name = coi_alien_scientist_extraction.1.c
		ai_chance = { factor = 20 }
		custom_effect_tooltip = coi_alien_scientist_extraction_removal_tt
		hidden_effect = {
			coi_alien_record_scientist_removed_effect = yes
			event_target:coi_alien_extraction_target = {
				add_timed_idea = {
					idea = coi_alien_scientific_leadership_disrupted
					days = 180
				}
				every_active_scientist = {
					limit = {
						NOT = { has_trait = coi_alien_scientist_breakthrough_disrupted }
					}
					add_scientist_trait = coi_alien_scientist_breakthrough_disrupted
				}
				# This hidden event removes the CHARACTER-category component at
				# the same 180-day horizon as the timed country idea.
				country_event = {
					id = coi_alien_scientist_extraction.5
					days = 180
				}
			}
		}
	}
}
