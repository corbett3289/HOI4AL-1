### Six-month resolution. Scores at or below forty stabilize. A score above
### seventy becomes existential only if stability is also below twenty-five.
country_event = {
	id = coi_alien_catastrophic_disclosure.90
	title = coi_alien_catastrophic_disclosure.90.t
	desc = coi_alien_catastrophic_disclosure.90.desc
	picture = GFX_report_event_coi_alien_catastrophic_disclosure
	is_triggered_only = yes
	trigger = { has_country_flag = coi_alien_disclosure_crisis_active }

	option = {
		name = coi_alien_catastrophic_disclosure.90.a
		trigger = { check_variable = { coi_alien_crisis_score < 41 } }
		coi_alien_disclosure_stabilized_effect = yes
	}
	option = {
		name = coi_alien_catastrophic_disclosure.90.b
		trigger = {
			check_variable = { coi_alien_crisis_score > 40 }
		OR = {
			check_variable = { coi_alien_crisis_score < 71 }
			NOT = { has_stability < 0.25 }
		}
		}
		coi_alien_disclosure_leadership_collapse_effect = yes
	}
	option = {
		name = coi_alien_catastrophic_disclosure.90.c
		trigger = {
			check_variable = { coi_alien_crisis_score > 70 }
			has_stability < 0.25
		}
		country_event = { id = coi_alien_catastrophic_disclosure.91 hours = 1 }
	}
}
