### Managed disclosure unfolds at sixty-day intervals. The persistent public
### variable is monotonic, while the stance check safely terminates an authored
### policy change or a future catastrophic-disclosure override.
country_event = {
	id = coi_alien_discovery.10
	title = coi_alien_discovery.10.t
	desc = coi_alien_discovery.10.desc
	picture = GFX_report_event_coi_alien_disclosure_managed
	is_triggered_only = yes
	trigger = {
		has_country_flag = coi_alien_disclosure_stance_managed
		has_country_flag = coi_alien_managed_disclosure_in_progress
		NOT = { has_global_flag = coi_alien_catastrophic_disclosure_active }
	}

	option = {
		name = coi_alien_discovery.10.a
		coi_alien_raise_public_awareness_to_rumors_effect = yes
		country_event = { id = coi_alien_discovery.11 days = 60 }
	}
}
